cmake_minimum_required(VERSION 3.18)
project(CRISP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_library(crisp_fss test/relu/fss_comparison.cpp test/relu/relu.cpp)
target_include_directories(crisp_fss PUBLIC test/relu)
target_link_libraries(crisp_fss PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_options(crisp_fss PRIVATE -Wall -Wextra)
add_executable(test_fss tests/test_fss.cpp)
target_link_libraries(test_fss PRIVATE crisp_fss)
enable_testing()
add_test(NAME fss_correctness COMMAND test_fss)
# Historical EMP harnesses are superseded: they send dummy keys and have invalid
# timer boundaries. They remain in test/ solely for source comparison.
add_executable(crisp_relu bench/relu_bench.cpp)
target_link_libraries(crisp_relu PRIVATE crisp_fss)
add_executable(crisp_pool bench/pool_bench.cpp)
target_link_libraries(crisp_pool PRIVATE crisp_fss)
