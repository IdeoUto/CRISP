cmake_minimum_required(VERSION 3.18)
project(CRISP_CKKS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SEAL 3.6 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
if(APPLE)
 find_path(OMP_INCLUDE omp.h HINTS /opt/homebrew/opt/libomp/include REQUIRED)
 find_library(OMP_LIBRARY omp HINTS /opt/homebrew/opt/libomp/lib REQUIRED)
endif()
add_library(crisp_ckks validated/kernels.cpp)
target_include_directories(crisp_ckks PUBLIC validated)
target_link_libraries(crisp_ckks PUBLIC SEAL::seal Threads::Threads OpenSSL::Crypto)
if(APPLE)
 target_compile_options(crisp_ckks PUBLIC -Xpreprocessor -fopenmp)
 target_include_directories(crisp_ckks PUBLIC ${OMP_INCLUDE})
 target_link_libraries(crisp_ckks PUBLIC ${OMP_LIBRARY})
else()
 find_package(OpenMP REQUIRED)
 target_link_libraries(crisp_ckks PUBLIC OpenMP::OpenMP_CXX)
endif()
enable_testing()
add_executable(test_ckks validated/test_kernels.cpp)
target_link_libraries(test_ckks PRIVATE crisp_ckks)
add_test(NAME ckks_correctness COMMAND test_ckks)
add_executable(crisp_linear validated/bench.cpp)
target_link_libraries(crisp_linear PRIVATE crisp_ckks)
set_target_properties(crisp_ckks PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(crisp_model_bridge SHARED validated/model_bridge.cpp ../CRISP/test/relu/fss_comparison.cpp ../CRISP/test/relu/relu.cpp)
target_link_libraries(crisp_model_bridge PRIVATE crisp_ckks)
option(CRISP_BUILD_SIMC_REFERENCE "Build explicitly named ring-32 SIMC reference" ON)
if(CRISP_BUILD_SIMC_REFERENCE)
 find_path(EMP_INCLUDE emp-tool/emp-tool.h HINTS /usr/local/include REQUIRED)
 find_library(EMP_LIBRARY emp-tool HINTS /usr/local/lib REQUIRED)
 add_executable(simc_ring32 validated/simc_ring32.cpp)
 target_include_directories(simc_ring32 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${EMP_INCLUDE})
 target_compile_definitions(simc_ring32 PRIVATE THREADING)
 if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  target_compile_options(simc_ring32 PRIVATE -march=armv8-a+simd+crypto+crc)
 endif()
 target_link_libraries(simc_ring32 PRIVATE SEAL::seal OpenSSL::Crypto OpenSSL::SSL Threads::Threads ${EMP_LIBRARY})
 if(APPLE)
  add_custom_command(TARGET simc_ring32 POST_BUILD COMMAND install_name_tool -change libemp-tool.dylib ${EMP_LIBRARY} $<TARGET_FILE:simc_ring32>)
 endif()
endif()
# Separate process prevents PyTorch's bundled OpenMP runtime from colliding
# with the Homebrew runtime used by SEAL kernels in the same address space.
add_executable(crisp_model_worker validated/model_worker.cpp)
target_link_libraries(crisp_model_worker PRIVATE crisp_model_bridge)
